## Regenerating this index

```js
const page = figma.root.children.find(p => p.name === 'Web V2'); // or 'Mobile V2'
await figma.setCurrentPageAsync(page);
const rows = [];
for (const s of page.children.filter(n => n.type === 'SECTION').sort((a, b) => a.y - b.y))
  for (const f of s.children.filter(n => n.type === 'FRAME')) rows.push(`| ${s.name.split(' ')[0]} | ${f.id} | ${f.name} |`);
return rows.join('\n');
```
